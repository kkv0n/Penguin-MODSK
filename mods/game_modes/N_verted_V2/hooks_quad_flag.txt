// PATCH LAP SKIPS ------------------------------------
// Track previous quadblocks for each player
struct QuadBlock* lastValidQuadblocks[4] = {NULL, NULL, NULL, NULL};
// Track quadblocks with temporary mask grab flags
static struct QuadBlock* flaggedQuadblocks[4] = {NULL, NULL, NULL, NULL};
static bool flagsApplied[4] = {false, false, false, false};

void ManualMaskGrab(struct Thread* t, struct Driver* d) {
    // Store original position values
    int origX = d->posCurr.x;
    int origY = d->posCurr.y;
    int origZ = d->posCurr.z;
    
    // Call original mask grab
    VehStuckProc_MaskGrab_Init(t, d);
    
    // Restore original position
    d->posCurr.x = origX;
    d->posCurr.y = origY;
    d->posCurr.z = origZ;
    
    // Fix previous position too (prevents teleporting on next frame)
    d->posPrev.x = origX;
    d->posPrev.y = origY;
    d->posPrev.z = origZ;
    
    // Position mask to follow player
    struct MaskHeadWeapon* mask = d->KartStates.MaskGrab.maskObj;
    if (mask != NULL) {
        mask->pos[0] = origX >> 8;
        mask->pos[1] = (origY >> 8) + 0x140; // Position mask above player
        mask->pos[2] = origZ >> 8;
    }
}

// Prevent lap skips (NMZ Abuse)
void preventLapSkip(struct Driver* driver, int driverIndex) {
    struct GameTracker *gGT = sdata->gGT;
    
    // Remove flag from previously modified quadblock for this driver
    if (flagsApplied[driverIndex] && flaggedQuadblocks[driverIndex] != NULL) {
        flaggedQuadblocks[driverIndex]->quadFlags &= ~Q_MASK_GRAB;
        flagsApplied[driverIndex] = false;
    }
    
    // Continue only if driver has a valid quadblock
    if (driver != NULL && driver->lastValid != NULL) {
        char maxCheckpoint = gGT->level1->cnt_restart_points - 1;
        char currentCheckpoint = driver->lastValid->checkpointIndex;
        
        // Only check if we have a previous valid checkpoint
        if (lastValidQuadblocks[driverIndex] != NULL && currentCheckpoint != lastValidQuadblocks[driverIndex]->checkpointIndex) {
            char prevCheckpoint = lastValidQuadblocks[driverIndex]->checkpointIndex;
            
            // Detect lap skip - from after checkpoint 5 to final checkpoint with big jump
            if (
                currentCheckpoint == maxCheckpoint
                && prevCheckpoint > 5
                && abs((int)currentCheckpoint - (int)prevCheckpoint) > 2
            ) {
                // Apply the mask grab flag to the quadblock
                driver->lastValid->quadFlags |= Q_MASK_GRAB;
                
                // Store which quadblock was modified for this driver
                flaggedQuadblocks[driverIndex] = driver->lastValid;
                flagsApplied[driverIndex] = true;
                
                // Reset to previous valid quadblock
                driver->lastValid = lastValidQuadblocks[driverIndex];
                
                // Visual feedback (only for human players)
                if ((driver->actionsFlagSet & 0x100000) == 0) {  // Check if not AI
                    sprintf(debugText, "LAP SKIP BLOCKED! P%d", driverIndex+1);
                    DecalFont_DrawLine(debugText, 0x100, 0xc8 + 10, FONT_SMALL, (JUSTIFY_CENTER | RED));
                }
            }
        }
        
        // Update previous valid quadblock for this driver
        lastValidQuadblocks[driverIndex] = driver->lastValid;
    }
}
